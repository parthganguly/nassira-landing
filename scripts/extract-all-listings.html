Extract All PropertyFinder Listings

Instructions:

  1. Open your PropertyFinder broker page
  2. Open Browser DevTools (F12 or Right-click → Inspect)
  3. Go to the Console tab
  4. Copy and paste the script below into the console
  5. Press Enter - it will extract all listings from the page
  6. Copy the JSON output and save it

Console Script:

// Extract all listings from PropertyFinder broker page (async function() { console.log('Extracting all listings...'); // Get initial data const script = document.getElementById('__NEXT_DATA__'); if (!script) { console.error('Could not find __NEXT_DATA__'); return; } const data = JSON.parse(script.textContent); const broker = data.props.pageProps.broker; const initialProps = data.props.pageProps.properties.data.properties || []; const meta = data.props.pageProps.properties.data.meta || {}; console.log(`Found ${initialProps.length} on page 1`); console.log(`Total according to meta: ${meta.total_count || 'unknown'}`); let allProperties = [...initialProps]; // Try to get more by checking if there are additional pages const totalCount = meta.total_count || initialProps.length; const pageCount = meta.page_count || 1; console.log(`Attempting to fetch ${pageCount} pages...`); // Scroll through and wait for dynamic content for (let i = 0; i < 10; i++) { window.scrollTo(0, document.body.scrollHeight); await new Promise(r => setTimeout(r, 1000)); // Re-extract after scroll const newScript = document.getElementById('__NEXT_DATA__'); if (newScript) { try { const newData = JSON.parse(newScript.textContent); const newProps = newData.props.pageProps.properties.data.properties || []; const existingIds = new Set(allProperties.map(p => p.id)); const unique = newProps.filter(p => !existingIds.has(p.id)); if (unique.length > 0) { allProperties.push(...unique); console.log(`Found ${unique.length} new properties (${allProperties.length} total)`); } } catch (e) {} } } // Format for our use const formatted = allProperties.map(prop => ({ id: parseInt(prop.id), type: (prop.property_type || 'PROPERTY').toUpperCase(), status: prop.price?.period === 'rent' ? 'FOR RENT' : 'FOR SALE', price: prop.price?.value ? `AED ${prop.price.value.toLocaleString()}` : 'Price on Request', title: prop.title?.trim() || 'Luxury Property', location: prop.location?.full_name || 'Dubai, UAE', beds: parseInt(prop.bedrooms) || 0, baths: parseInt(prop.bathrooms) || 0, sqft: prop.size?.value || 0, image: prop.images?.[0]?.medium || prop.images?.[0]?.small || '/placeholder.svg?height=400&width=600' })); console.log(`\\n✅ Total listings extracted: ${formatted.length}`); console.log('\\n📋 Copy the JSON below:'); console.log('\\n' + '='.repeat(60)); console.log(JSON.stringify(formatted, null, 2)); console.log('='.repeat(60)); return formatted; })();
